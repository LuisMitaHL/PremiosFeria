-- 1. Firmar códigos QR en el servidor (el secreto nunca sale de la base de datos)
-- Nota de despliegue: rota el secreto UNA vez tras aplicar esta versión,
-- porque el valor anterior fue público (hallazgo F3 del audit):
--   UPDATE settings SET value = gen_random_uuid()::text || gen_random_uuid()::text
--   WHERE key = 'hmac_secret';
DROP FUNCTION IF EXISTS simple_hmac(TEXT, TEXT);

CREATE OR REPLACE FUNCTION sign_scan_code(
  p_community_id UUID,
  p_type TEXT
) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE
  v_comm communities%ROWTYPE;
  v_secret TEXT;
  v_ts BIGINT;
  v_pts INT;
  v_tok TEXT;
BEGIN
  IF p_type NOT IN ('visit', 'activity') THEN
    RETURN jsonb_build_object('error', 'Tipo inválido');
  END IF;

  SELECT * INTO v_comm FROM communities WHERE id = p_community_id;
  IF NOT FOUND THEN
    RETURN jsonb_build_object('error', 'Comunidad no encontrada');
  END IF;

  IF v_comm.auth_user_id IS DISTINCT FROM auth.uid() THEN
    RETURN jsonb_build_object('error', 'No autorizado');
  END IF;

  SELECT value INTO v_secret FROM settings WHERE key = 'hmac_secret';
  IF v_secret IS NULL THEN
    RETURN jsonb_build_object('error', 'Secreto no configurado');
  END IF;

  v_ts := EXTRACT(EPOCH FROM NOW())::BIGINT / 15;
  -- Los puntos nunca son negativos (la configuración del stand podría serlo)
  v_pts := GREATEST(0, CASE WHEN p_type = 'visit'
    THEN LEAST(COALESCE(v_comm.visit_points, 10), 30)
    ELSE LEAST(COALESCE(v_comm.activity_points, 25), 100)
  END);

  v_tok := encode(
    hmac(v_comm.id || '|' || v_ts || '|' || v_pts || '|' || p_type, v_secret, 'sha256'),
    'hex'
  );

  RETURN jsonb_build_object(
    'payload', jsonb_build_object(
      'sid', v_comm.id,
      'ts', v_ts,
      'pts', v_pts,
      'type', p_type,
      'name', v_comm.name,
      'emoji', v_comm.emoji,
      'tok', v_tok
    ),
    'shortCode', upper(substring(v_tok from 1 for 6)),
    'ts', v_ts
  );
END;
$$;

-- 2. Validar QR y registrar scan (Lógica de validación temporal y de puntos 100% segura en el backend)
CREATE OR REPLACE FUNCTION validate_and_scan(
  p_encoded_payload TEXT
) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE
  v_json JSONB;
  v_secret TEXT;
  v_community communities%ROWTYPE;
  v_existing scans%ROWTYPE;
  v_elapsed INTERVAL;
  
  -- Para validaciones y limites:
  v_type TEXT;
  v_pts INT;
  v_qr_ts BIGINT;
  v_current_ts BIGINT;
  v_max_pts INT;
  v_final_pts INT;
  v_expected_tok TEXT;
  v_participant_id UUID;
BEGIN
  -- El participante es SIEMPRE el llamador autenticado (nunca un parámetro)
  SELECT id INTO v_participant_id FROM participants WHERE auth_user_id = auth.uid();
  IF NOT FOUND THEN
    RETURN jsonb_build_object('valid', false, 'reason', '👤 Regístrate para participar.');
  END IF;
  -- Decodificar payload (El frontend debe enviar el JSON parseable como string antes de insertarlo)
  v_json := p_encoded_payload::JSONB;
  -- ==========================================
  -- 0) SELECCIONAR FLUJO (CÓDIGO MANUAL vs QR NORMAL)
  -- ==========================================
  IF v_json ? 'short_code' THEN
    DECLARE
      v_short_code TEXT := upper(v_json->>'short_code');
      v_comp_comm communities%ROWTYPE;
      v_comp_type TEXT;
      v_comp_pts INT;
      v_found BOOLEAN := false;
    BEGIN
      SELECT value INTO v_secret FROM settings WHERE key = 'hmac_secret';
      v_current_ts := EXTRACT(EPOCH FROM NOW())::BIGINT / 15;

      -- Buscar entre todas las comunidades la firma que coincida con este código
      FOR v_comp_comm IN SELECT * FROM communities LOOP
        FOR v_comp_type IN SELECT unnest(ARRAY['visit', 'activity']) LOOP
          v_comp_pts := CASE WHEN v_comp_type = 'visit' THEN GREATEST(0, LEAST(COALESCE(v_comp_comm.visit_points, 10), 30)) ELSE GREATEST(0, LEAST(COALESCE(v_comp_comm.activity_points, 25), 100)) END;
          
          -- Intentar con el TS actual
          IF upper(substring(encode(hmac(v_comp_comm.id || '|' || v_current_ts || '|' || v_comp_pts || '|' || v_comp_type, v_secret, 'sha256'), 'hex') from 1 for 6)) = v_short_code THEN
            v_community := v_comp_comm;
            v_type := v_comp_type;
            v_pts := v_comp_pts;
            v_qr_ts := v_current_ts;
            v_found := true;
            EXIT;
          END IF;
          
          -- Intentar con el TS anterior (tolerancia de red)
          IF upper(substring(encode(hmac(v_comp_comm.id || '|' || (v_current_ts - 1) || '|' || v_comp_pts || '|' || v_comp_type, v_secret, 'sha256'), 'hex') from 1 for 6)) = v_short_code THEN
            v_community := v_comp_comm;
            v_type := v_comp_type;
            v_pts := v_comp_pts;
            v_qr_ts := v_current_ts - 1;
            v_found := true;
            EXIT;
          END IF;
        END LOOP;
        EXIT WHEN v_found;
      END LOOP;

      IF NOT v_found THEN
        RETURN jsonb_build_object('valid', false, 'reason', '❌ Código manual inválido o caducado.');
      END IF;
    END;

  ELSE
    -- ==========================================
    -- FLUJO NORMAL DE QR (JSON PAYLOAD BASE64)
    -- ==========================================
    v_type := COALESCE(v_json->>'type', 'visit');
    v_pts := (v_json->>'pts')::INT;

    -- A) VALIDAR TIEMPO DEL CÓDIGO QR (Regeneración de 15 segundos)
    v_qr_ts := (v_json->>'ts')::BIGINT;
    v_current_ts := EXTRACT(EPOCH FROM NOW())::BIGINT / 15;
    
    -- Permitir tolerancia de 1 ventana hacia atrás por posible delay de red
    IF (v_current_ts - v_qr_ts) > 1 OR (v_qr_ts > v_current_ts) THEN
      RETURN jsonb_build_object('valid', false, 'reason', '⏱️ Código QR caducado, escanee el código actual del stand.');
    END IF;

    -- B) VERIFICAR LA COMUNIDAD Y LA INTEGRIDAD CRIPTOGRÁFICA
    SELECT * INTO v_community FROM communities WHERE id = (v_json->>'sid')::UUID;
    IF NOT FOUND THEN
      RETURN jsonb_build_object('valid', false, 'reason', '❌ Comunidad no encontrada');
    END IF;

    SELECT value INTO v_secret FROM settings WHERE key = 'hmac_secret';
    
    -- Verificar si el token (HMAC) no fue alterado
    v_expected_tok := encode(hmac(v_community.id || '|' || v_qr_ts || '|' || v_pts || '|' || v_type, v_secret, 'sha256'), 'hex');
    IF v_expected_tok IS DISTINCT FROM (v_json->>'tok') THEN
      RETURN jsonb_build_object('valid', false, 'reason', '🛡️ Código QR inválido o falsificado.');
    END IF;
  END IF;

  -- ==========================================
  -- C) VERIFICAR COOLDOWN / UNICIDAD
  --    Visitas: repetibles tras 5 minutos. Actividades: una sola vez
  --    por stand (respaldado por un índice único parcial en scans).
  -- ==========================================
  IF v_type = 'visit' THEN
    SELECT * INTO v_existing FROM scans
      WHERE participant_id = v_participant_id
      AND community_id = v_community.id
      AND type = 'visit'
      ORDER BY created_at DESC LIMIT 1;

    IF FOUND THEN
      v_elapsed := now() - v_existing.created_at;
      IF v_elapsed < INTERVAL '5 minutes' THEN
        RETURN jsonb_build_object('valid', false, 'reason',
          '🔒 Ya visitaste este stand. Espera ' ||
          CEIL(EXTRACT(EPOCH FROM (INTERVAL '5 minutes' - v_elapsed)) / 60)::TEXT ||
          ' minuto(s) para volver a registrar una visita.');
      END IF;
    END IF;
  ELSE
    SELECT * INTO v_existing FROM scans
      WHERE participant_id = v_participant_id
      AND community_id = v_community.id
      AND type = 'activity'
      LIMIT 1;

    IF FOUND THEN
      RETURN jsonb_build_object('valid', false, 'reason',
        '✅ Ya registraste esta actividad en este stand.');
    END IF;
  END IF;

  -- ==========================================
  -- D) VALIDAR LÍMITES DE PUNTOS DE LA BASE DE DATOS
  -- ==========================================
  IF v_type = 'visit' THEN
    v_max_pts := LEAST(COALESCE(v_community.visit_points, 10), 30);
  ELSE
    v_max_pts := LEAST(COALESCE(v_community.activity_points, 25), 100);
  END IF;
  
  -- Aunque el frontend envíe algo más grande (hackeado), usamos el límite máximo forzado por la BD:
  -- Aunque el payload traiga puntos negativos, el crédito nunca resta
  v_final_pts := GREATEST(0, LEAST(v_pts, v_max_pts));
  -- El índice único parcial es la puerta real contra la carrera de
  -- actividades concurrentes; aquí solo traducimos el rechazo.
  BEGIN
    INSERT INTO scans (participant_id, community_id, points, type)
    VALUES (v_participant_id, v_community.id, v_final_pts, v_type);
  EXCEPTION
    WHEN unique_violation THEN
      RETURN jsonb_build_object('valid', false, 'reason',
        '✅ Ya registraste esta actividad en este stand.');
  END;

  UPDATE participants SET points = points + v_final_pts
  WHERE id = v_participant_id;

  RETURN jsonb_build_object(
    'valid', true,
    'points', v_final_pts,
    'type', v_type,
    'groupName', v_community.name,
    'groupEmoji', v_community.emoji
  );
END;
$$;

-- 3. Reclamar premio
CREATE OR REPLACE FUNCTION claim_reward(
  p_reward_id UUID
) RETURNS JSONB LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$
DECLARE
  v_participant participants%ROWTYPE;
  v_reward rewards%ROWTYPE;
  v_already_claimed BOOLEAN;
  v_new_points INT;
BEGIN
  -- El participante es SIEMPRE el llamador autenticado (nunca un parámetro)
  SELECT * INTO v_participant FROM participants WHERE auth_user_id = auth.uid();
  IF NOT FOUND THEN RETURN jsonb_build_object('success', false, 'reason', 'No registrado'); END IF;

  SELECT * INTO v_reward FROM rewards WHERE id = p_reward_id;
  IF NOT FOUND THEN RETURN jsonb_build_object('success', false, 'reason', 'Premio no encontrado'); END IF;

  SELECT EXISTS(SELECT 1 FROM claimed_rewards WHERE participant_id = v_participant.id AND reward_id = p_reward_id) INTO v_already_claimed;
  IF v_already_claimed THEN RETURN jsonb_build_object('success', false, 'reason', 'Ya reclamaste este premio'); END IF;

  IF v_participant.points < v_reward.cost THEN
    RETURN jsonb_build_object('success', false, 'reason', 'Necesitas ' || (v_reward.cost - v_participant.points)::TEXT || ' puntos más');
  END IF;

  IF v_reward.stock <= 0 THEN
    RETURN jsonb_build_object('success', false, 'reason', '¡Agotado! Ya no quedan unidades de este premio');
  END IF;

  -- Los pre-chequeos de arriba solo dan mensajes amigables; la puerta real
  -- son los UPDATE condicionales: la condición vive en el WHERE, así que dos
  -- reclamaciones concurrentes se serializan en el lock de la fila y la
  -- perdedora re-evalúa contra la fila ya confirmada y ve 0 filas.
  -- El bloque con EXCEPTION es una subtransacción: si algo falla dentro
  -- (RAISE o error de BD), se revierte TODO lo escrito en él — nunca queda
  -- stock descontado sin puntos debitados, ni viceversa.
  BEGIN
    UPDATE rewards SET stock = stock - 1
    WHERE id = p_reward_id AND stock > 0;
    IF NOT FOUND THEN
      RAISE EXCEPTION '%', '¡Agotado! Ya no quedan unidades de este premio';
    END IF;

    UPDATE participants SET points = points - v_reward.cost
    WHERE id = v_participant.id AND points >= v_reward.cost;
    IF NOT FOUND THEN
      -- Tu balance cambió por una transacción concurrente entre el pre-chequeo y aquí
      RAISE EXCEPTION '%', 'Tus puntos cambiaron, inténtalo de nuevo';
    END IF;

    INSERT INTO claimed_rewards (participant_id, reward_id) VALUES (v_participant.id, p_reward_id);
  EXCEPTION
    WHEN unique_violation THEN
      -- Carrera de la misma persona contra el UNIQUE(participant_id, reward_id)
      RETURN jsonb_build_object('success', false, 'reason', 'Ya reclamaste este premio');
    WHEN OTHERS THEN
      RETURN jsonb_build_object('success', false, 'reason', SQLERRM);
  END;

  SELECT points INTO v_new_points FROM participants WHERE id = v_participant.id;
  RETURN jsonb_build_object('success', true, 'newPoints', v_new_points);
END;
$$;
